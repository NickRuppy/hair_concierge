'use strict';
// Filesystem/TypeScript parser only. Never imports a test or product owner, spawns a child, or writes the checkout.
const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto');
const D=__dirname,initial=JSON.parse(fs.readFileSync(path.join(D,'start-pins.json'),'utf8')),ROOT=initial.root;
const ts=require(path.join(ROOT,'node_modules/typescript/lib/typescript.js'));
const sha=x=>crypto.createHash('sha256').update(x).digest('hex');
const read=x=>fs.readFileSync(x,'utf8');const json=x=>JSON.parse(read(path.join(D,x)));const checks=[];
function ok(value,label){if(!value)throw Error(label);checks.push(label);}
function inventory(file,source){const sf=ts.createSourceFile(file,source,ts.ScriptTarget.Latest,true,ts.ScriptKind.TS);ok(sf.parseDiagnostics.length===0,'parse '+file);const rows=[];
 function walk(n){if(ts.isCallExpression(n)&&n.expression.getText(sf)==='test'){const cb=n.arguments.find(x=>ts.isArrowFunction(x)||ts.isFunctionExpression(x));if(cb){const calls=[],assertions=[];function inner(x){if(ts.isCallExpression(x)){const expression=x.expression.getText(sf),item={expression,text:x.getText(sf),line:sf.getLineAndCharacterOfPosition(x.getStart(sf)).line+1};calls.push(item);if(expression.startsWith('assert.'))assertions.push(item);}ts.forEachChild(x,inner)}inner(cb);rows.push({title:n.arguments[0].getText(sf),line:sf.getLineAndCharacterOfPosition(n.getStart(sf)).line+1,callback:cb.getText(sf),callbackSHA256:sha(cb.getText(sf)),calls,assertions});}}ts.forEachChild(n,walk)}walk(sf);return rows;}
for(const p of initial.pins)ok(sha(fs.readFileSync(path.join(ROOT,p.file)))===p.sha256,'initial pin '+p.file);
const phases={};for(const phase of ['before','transfer','cut']){phases[phase]=[];for(const file of initial.testFiles){const source=read(path.join(D,'snapshots',phase,file));phases[phase].push({file,sha256:sha(source),rows:inventory(file,source)});}}
const candidate=json('candidates.json')[0], file=candidate.file;
const group=phase=>phases[phase].find(x=>x.file===file), keeper=phase=>group(phase).rows.find(x=>x.title===candidate.keeper.title);
const donor=group('before').rows.find(x=>x.title===candidate.donor.title),originalKeeper=keeper('before');
ok(phases.before.reduce((n,f)=>n+f.rows.length,0)===102,'before102');ok(phases.transfer.reduce((n,f)=>n+f.rows.length,0)===102,'transfer102');ok(phases.cut.reduce((n,f)=>n+f.rows.length,0)===101,'cut101');
let untouched=0;for(const f of phases.before)for(const row of f.rows){if(f.file===file&&(row.title===candidate.donor.title||row.title===candidate.keeper.title))continue;for(const phase of ['transfer','cut']){const found=phases[phase].find(x=>x.file===f.file).rows.find(x=>x.title===row.title);ok(found&&found.callbackSHA256===row.callbackSHA256,'unrelated '+phase+' '+f.file+':'+row.line)}untouched++}ok(untouched===100,'100 unrelated exact');
ok(group('transfer').rows.find(x=>x.title===donor.title).callbackSHA256===donor.callbackSHA256,'donor retained exact in transfer');ok(!group('cut').rows.some(x=>x.title===donor.title),'only donor absent');ok(keeper('transfer').callbackSHA256===keeper('cut').callbackSHA256,'transfer cut same keeper');
ok(donor.assertions.length===7,'seven donor assertions');for(const phase of ['transfer','cut']){const k=keeper(phase);ok(k.assertions.length===originalKeeper.assertions.length+donor.assertions.length,'full assertion count '+phase);for(let i=0;i<originalKeeper.assertions.length;i++)ok(k.assertions[i].text===originalKeeper.assertions[i].text,'original oracle '+phase+' '+i);for(let i=0;i<donor.assertions.length;i++)ok(k.assertions[i+originalKeeper.assertions.length].text===donor.assertions[i].text,'donor oracle byte exact '+phase+' '+i);const nonAssertionOwner=k.calls.filter(c=>c.expression==='compileInitialRoutineCandidate');ok(nonAssertionOwner.length===2&&nonAssertionOwner.every((c,i)=>c.text===originalKeeper.calls.filter(x=>x.expression==='compileInitialRoutineCandidate')[i].text),'two existing owner inputs exact '+phase);}
const controls=json('controls.json');for(const c of controls){const source=read(path.join(D,'originals',c.source));ok(sha(source)===c.originalSourceSHA256,'control owner '+c.id);ok(source.split(c.find).length-1===1,'unique control anchor '+c.id);const mutant=source.replace(c.find,c.replace);ok(sha(mutant)===c.mutantSHA256,'mutant hash '+c.id);const sf=ts.createSourceFile(c.source,mutant,ts.ScriptTarget.Latest,true);ok(sf.parseDiagnostics.length===0,'control parse '+c.id);for(const phase of ['transfer','cut']){const k=keeper(phase),a=k.assertions[originalKeeper.assertions.length+c.oracle.transferredAssertionOrdinal-1];c.oracle[phase]={line:a.line,assertion:a.text,keeperRegistrationLine:k.line,firstFrameFile:file};}}
fs.writeFileSync(path.join(D,'controls.json'),JSON.stringify(controls,null,2)+'\n');fs.writeFileSync(path.join(D,'phase-inventory.json'),JSON.stringify(phases,null,2)+'\n');
const result={status:'STATIC PASS ONLY',counts:{before:102,transfer:102,cut:101,unrelatedExact:100,controlDescriptors:controls.length},checks,semanticLimits:['No runtime, native assertion, typecheck, baseline or source-fault result is claimed.','Parser loads TypeScript only.','Full scope test bodies include17 recovered current-byte product-preview reviews; full fresh review85.','Catalog asset snapshots/consumed structural projections are not full human provenance/copy audit.']};fs.writeFileSync(path.join(D,'static-check.json'),JSON.stringify(result,null,2)+'\n');process.stdout.write(JSON.stringify({status:result.status,counts:result.counts,checks:checks.length})+'\n');
