'use strict';
// Static in-memory diagnostic-parser checks only. No owner/test execution or source writes.
const fs=require('node:fs'),proof=require('./operator-proof.cjs'),controls=JSON.parse(fs.readFileSync(__dirname+'/operator-controls-normalized.json','utf8'));
let assertions=0;function ok(value,message){assertions++;if(!value)throw Error(message)}
function tap(c,phase){const o=c.oracles[phase];return `TAP version 13\n# Subtest: ${c.test.name}\nnot ok 1 - ${c.test.name}\n  ---\n  error: |-\n    ${c.errorMessageIncludes.join('\n').replaceAll('\n','\n    ')}\n  code: 'ERR_ASSERTION'\n  name: 'AssertionError'\n  operator: '${c.expectedOperator}'\n  stack: |-\n    TestContext.<anonymous> (${c.test.file}:${o.startLine}:10)\n    Test.run (node:internal/test_runner/test:1000:1)\n  ...\n1..1\n# tests 1\n# pass 0\n# fail 1\n# cancelled 0\n# skipped 0\n# todo 0\n`}
function run(text,extra={}){return{status:1,signal:null,timedOut:false,spawnError:null,...proof.parseTAP(text),text,...extra}}
for(const c of controls)for(const phase of ['transfer','cut']){
 const text=tap(c,phase);ok(proof.intended(c,run(text),phase).accepted,c.id+' intended');
 ok(!proof.intended(c,run(text,{timedOut:true}),phase).accepted,'timeout rejected');
 ok(!proof.intended(c,run(text,{signal:'SIGKILL'}),phase).accepted,'signal rejected');
 ok(!proof.intended(c,run(text,{spawnError:'error'}),phase).accepted,'spawn rejected');
 ok(!proof.intended(c,run(text.replace("code: 'ERR_ASSERTION'","code: 'ERR_MODULE_NOT_FOUND'")),phase).accepted,'import rejected');
 ok(!proof.intended(c,run(text.replace("operator: '"+c.expectedOperator+"'","operator: '!='")),phase).accepted,'operator rejected');
 ok(!proof.intended(c,run(text.replace(c.test.file+':'+c.oracles[phase].startLine+':10',c.test.file+':1:10')),phase).accepted,'first frame rejected');
 ok(!proof.intended(c,run(text.replace('# tests 1','# tests 0')),phase).accepted,'zero rejected');
 ok(!proof.intended(c,run(text.replace('# skipped 0','# skipped 1')),phase).accepted,'skip rejected');
 ok(!proof.intended(c,run(text.replace('# Subtest: '+c.test.name,'# Subtest: Wrong test')),phase).accepted,'wrong name rejected');
 for(const part of c.errorMessageIncludes)ok(!proof.intended(c,run(text.replace(part.replaceAll('\n','\n    '),'unrelated diagnostic')),phase).accepted,'required message fragment rejected');
 const body=c.errorMessageIncludes.join(' | ');
 for(const scalar of [JSON.stringify(body), ...(!body.includes("\n")?["'"+body.replaceAll("'", "''")+"'"]:[])]){
  const scalarText=text.replace(/  error: \|-\n(?:    [^\n]*\n)+?(?=  code:)/,'  error: '+scalar+'\n');
  ok(proof.intended(c,run(scalarText),phase).accepted,c.id+' decoded scalar accepted');
 }
 ok(!proof.intended(c,run(text.replace('  error: |-','  error: >-')),phase).accepted,'unknown scalar rejected');
 const nestedWrong=text.replace("  operator: '"+c.expectedOperator+"'","  actual:\n    stack: |-\n      setup (src/unrelated.ts:1:1)\n  operator: '"+c.expectedOperator+"'");
 ok(proof.intended(c,run(nestedWrong),phase).accepted,'nested actual stack ignored');
 const wrongOuter=nestedWrong.replace(c.test.file+':'+c.oracles[phase].startLine+':10',c.test.file+':1:10').replace('setup (src/unrelated.ts:1:1)','TestContext.<anonymous> ('+c.test.file+':'+c.oracles[phase].startLine+':10)');
 ok(!proof.intended(c,run(wrongOuter),phase).accepted,'correct nested frame cannot rescue wrong assertion frame');

 const cleanText=`TAP version 13\n# Subtest: ${c.test.name}\nok 1 - ${c.test.name}\n1..1\n# tests 1\n# pass 1\n# fail 0\n# cancelled 0\n# skipped 0\n# todo 0\n`;
 ok(proof.clean(c,run(cleanText,{status:0})),'clean accepted');
 ok(!proof.clean(c,run(cleanText.replace('# tests 1','# tests 0'),{status:0})),'zero clean rejected');
}
console.log(JSON.stringify({status:'PASS_STATIC_SYNTHETIC_PARSER_ONLY',controls:5,phases:2,assertions,children:0,ownerImports:0,repositoryWrites:0,runtimeClaims:0,note:'Diagnostic-shape rejection checks only, not native fault sensitivity or real TAP evidence.'},null,2));
