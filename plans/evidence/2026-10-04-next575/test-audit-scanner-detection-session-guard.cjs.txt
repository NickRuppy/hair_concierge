const sha=b=>crypto.createHash('sha256').update(b).digest('hex')
const fail=message=>{throw Error(message)}
function checkFile(pin){const exists=fs.existsSync(pin.path);if(pin.sha===null){if(exists)fail('Previously absent identity file appeared: '+pin.path);return}if(!exists||sha(fs.readFileSync(pin.path))!==pin.sha||fs.realpathSync(pin.path)!==pin.realpath)fail('Identity/dependency drift: '+pin.path)}
function identity(pins){
 if(process.cwd()!==ROOT||fs.realpathSync(process.cwd())!==ROOT||pins.root!==ROOT)fail('Exact real checkout required')
 if(process.version!==pins.node.version||fs.realpathSync(process.execPath)!==pins.node.realpath||sha(fs.readFileSync(process.execPath))!==pins.node.sha)fail('Pinned Node binary required')
 for(const pin of pins.identityFiles)checkFile(pin)
 // Pinned actual Git metadata resolves this branch and exact HEAD without spawning git.
 const head=fs.readFileSync(path.join(pins.gitDir,'HEAD'),'utf8').trim()
 if(head!=='ref: refs/heads/'+pins.branch)fail('Branch drift')
 const ref=path.join(pins.commonDir,'refs/heads',pins.branch)
 const commit=fs.existsSync(ref)?fs.readFileSync(ref,'utf8').trim():fs.readFileSync(path.join(pins.commonDir,'packed-refs'),'utf8').split('\n').find(l=>l.endsWith(' refs/heads/'+pins.branch))?.split(' ')[0]
 if(commit!==pins.head)fail('HEAD drift')
}
function parse(ts,file,text){const t=ts.createSourceFile(file,text,ts.ScriptTarget.Latest,true,file.endsWith('.tsx')?ts.ScriptKind.TSX:ts.ScriptKind.TS);if(t.parseDiagnostics.length)fail('Parse: '+file);return t}
function registrations(ts,file,text){
 const sf=parse(ts,file,text),rows=new Map()
 for(const n of sf.statements){if(!ts.isExpressionStatement(n)||!ts.isCallExpression(n.expression)||n.expression.expression.getText(sf)!=='test')continue
 const call=n.expression,cb=call.arguments.at(-1),name=call.arguments[0].text
 if(typeof name!=='string'||!ts.isArrowFunction(cb)||!cb.body||rows.has(name))fail('Unexpected registration: '+file)
 const assertions=[],calls=[]
 function visit(x){if(ts.isCallExpression(x)){if(x.expression.getText(sf).startsWith('assert.'))assertions.push({startLine:sf.getLineAndCharacterOfPosition(x.getStart(sf)).line+1,text:x.getText(sf)});else calls.push(x.getText(sf))}ts.forEachChild(x,visit)}visit(cb)
 rows.set(name,{name,statement:n.getText(sf),start:n.getStart(sf),end:n.end,callback:cb.getText(sf),body:cb.body.getText(sf),assertions,calls:calls.sort()})
 }
 return rows
}
function replaceOnce(text,from,to,label){if(text.split(from).length!==2)fail('Non-unique anchor: '+label);return text.replace(from,()=>to)}
function support(text,rows){for(const r of [...rows.values()].reverse())text=text.slice(0,r.start)+text.slice(r.end);return text}
function load(pinSHA){
 const raw=fs.readFileSync(BASE+'-editor-pins.json');if(sha(raw)!==pinSHA)fail('Pins changed');const pins=JSON.parse(raw);identity(pins)
 for(const pin of pins.dependencies)checkFile(pin)
 for(const [file,hash]of Object.entries(pins.artifactHashes))if(sha(fs.readFileSync(file))!==hash)fail('Artifact changed: '+file)
 for(const[file,hash]of Object.entries(pins.entrypointTemplateHashes)){const s=fs.readFileSync(file,'utf8').replace(/const PIN_SHA='[a-f0-9]{64}'/,"const PIN_SHA='PIN_SHA_PLACEHOLDER'");if(sha(s)!==hash)fail('Operator code changed: '+file)}
 const evidence=JSON.parse(fs.readFileSync(BASE+'-evidence.json')),candidates=JSON.parse(fs.readFileSync(BASE+'-candidates.json')),controls=JSON.parse(fs.readFileSync(BASE+'-controls-normalized.json')),ts=require(ROOT+'/node_modules/typescript')
 if(evidence.root!==ROOT||evidence.head!==pins.head||candidates.length!==9||controls.length!==9||ts.version!==pins.typescriptVersion)fail('Identity/count mismatch')
 const snapshots={},counts={before:0,transfer:0,cut:0};let untouched=0
 for(const f of evidence.cohort){snapshots[f.file]={};for(const phase of Object.keys(counts)){const entry=f.phases[phase],bytes=Buffer.from(entry.text),rows=registrations(ts,f.file,entry.text);if(sha(bytes)!==entry.sha256||rows.size!==entry.declarations.length)fail('Phase hash/count');for(const r of entry.declarations){const actual=rows.get(r.title);if(!actual||sha(actual.statement)!==r.statementSha256||sha(actual.body)!==r.bodySha256)fail('Callback bytes mismatch')}snapshots[f.file][phase]={bytes,rows};counts[phase]+=rows.size}
 const {before,transfer,cut}=snapshots[f.file];let reconstructed=before.bytes.toString();const changed=new Map()
 for(const c of candidates.filter(c=>c.keeperFile===f.file)){const row=before.rows.get(c.keeper.title);if(!row||row.statement!==c.keeper.statement)fail('Keeper origin');if(changed.has(row.name))fail('Unexpected multiple keeper union');changed.set(row.name,c.keeperTransfer.statement);reconstructed=replaceOnce(reconstructed,row.statement,c.keeperTransfer.statement,c.id)}
 if(reconstructed!==transfer.bytes.toString())fail('Transfer reconstruction: '+f.file)
 for(const c of candidates.filter(c=>c.donorFile===f.file)){if(before.rows.get(c.donor.title)?.statement!==c.donor.statement)fail('Donor origin');reconstructed=replaceOnce(reconstructed,c.donor.statement,'',c.id)}
 if(reconstructed!==cut.bytes.toString())fail('Cut reconstruction: '+f.file)
 if(support(before.bytes.toString(),before.rows)!==support(transfer.bytes.toString(),transfer.rows)||support(before.bytes.toString(),before.rows)!==support(cut.bytes.toString(),cut.rows))fail('Support byte drift')
 for(const[name,row]of before.rows){const donor=candidates.some(c=>c.donorFile===f.file&&c.donor.title===name),transferred=transfer.rows.get(name);if(JSON.stringify(row.calls)!==JSON.stringify(transferred.calls))fail('Non-assertion input/call drift: '+name);if(!changed.has(name)&&transferred.statement!==row.statement)fail('Unrelated transfer change');if(donor){if(cut.rows.has(name))fail('Donor survived')}else{const retained=cut.rows.get(name);if(retained?.statement!==transferred.statement)fail('Retained bytes changed');if(retained.statement===row.statement)untouched++}}
 }
 if(Object.keys(snapshots).length!==4||JSON.stringify(counts)!=='{"before":152,"transfer":152,"cut":143}'||untouched!==137)fail('Conservation mismatch')
 function current(overrides={}){identity(pins);for(const pin of pins.dependencies)checkFile(pin);for(const r of pins.readset)if(sha(fs.readFileSync(path.join(ROOT,r.file)))!==(overrides[r.file]||r.sha))fail('Readset drift/fault active: '+r.file);const bytes=Object.fromEntries(Object.keys(snapshots).map(f=>[f,fs.readFileSync(path.join(ROOT,f))]));const phases=Object.keys(counts).filter(p=>Object.entries(bytes).every(([f,b])=>b.equals(snapshots[f][p].bytes)));if(phases.length!==1)fail('Mixed/unrecognized phase');return{phase:phases[0],bytes}}
 return{pins,evidence,candidates,controls,ts,snapshots,counts,untouched,current}
}
function fault(ctx,c){const original=fs.readFileSync(path.join(ROOT,c.file));if(sha(original)!==c.sourceSha256)fail('Fault owner drift: '+c.id);const text=original.toString(),t=parse(ctx.ts,c.file,text),found=[];function walk(n){if(ctx.ts.isFunctionDeclaration(n)&&n.name?.text===c.anchor.scope)found.push(n);ctx.ts.forEachChild(n,walk)}walk(t);if(found.length!==1||c.anchor.kind!=='function'||c.anchor.expectedOccurrences!==1)fail('Unique function: '+c.id);const n=found[0],mutated=Buffer.from(text.slice(0,n.getStart(t))+replaceOnce(n.getText(t),c.from,c.to,c.id)+text.slice(n.end));parse(ctx.ts,c.file,mutated.toString());if(sha(mutated)!==c.mutatedSha256)fail('Fault hash mismatch');for(const phase of['transfer','cut']){const row=ctx.snapshots[c.test.file][phase].rows.get(c.test.name),o=c.oracles[phase];if(!row?.assertions.some(a=>a.startLine===o.startLine&&a.text===o.text))fail('Oracle drift: '+c.id+'/'+phase)}return{original,mutated}}
function atomicCAS(file,expected,next){const target=path.join(ROOT,file),temp=target+'.test-audit-'+process.pid+'-'+crypto.randomUUID()+'.tmp';if(!fs.readFileSync(target).equals(expected))fail('Owned CAS refused: '+file);try{fs.writeFileSync(temp,next,{flag:'wx',mode:fs.statSync(target).mode});if(!fs.readFileSync(target).equals(expected))fail('Pre-rename CAS refused: '+file);fs.renameSync(temp,target)}finally{if(fs.existsSync(temp))fs.unlinkSync(temp)}}
function lock(){const token=crypto.randomUUID();fs.mkdirSync(LOCK);try{fs.writeFileSync(path.join(LOCK,'owner.json'),JSON.stringify({token,pid:process.pid,root:ROOT,operator:BASE}),{flag:'wx'})}catch(e){throw Error('New lock could not record owner; preserve lock: '+e)}return()=>{const owner=JSON.parse(fs.readFileSync(path.join(LOCK,'owner.json')));if(owner.token!==token)fail('Lock ownership changed; preserve lock');fs.unlinkSync(path.join(LOCK,'owner.json'));fs.rmdirSync(LOCK)}}
