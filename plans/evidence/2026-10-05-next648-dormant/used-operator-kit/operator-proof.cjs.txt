'use strict'
// Shared read-only TAP proof validation. Cut revalidates logs instead of trusting a boolean.
const fs=require('node:fs'),path=require('node:path')
const g=require('./operator-guard.cjs')
function parseTAP(text){
 const number=key=>{const all=[...text.matchAll(new RegExp('^# '+key+' (\\d+)$','gm'))];return all.length===1?Number(all[0][1]):null}
 return {tests:number('tests'),pass:number('pass'),fail:number('fail'),skipped:number('skipped'),cancelled:number('cancelled'),todo:number('todo')}
}
function named(control,run){const names=[...run.text.matchAll(/^# Subtest: (.+)$/gm)].map(m=>m[1]);return names.length===1&&names[0]===control.test.name}
function clean(control,run){return run.status===0&&run.signal===null&&!run.timedOut&&!run.spawnError&&run.tests===1&&run.pass===1&&run.fail===0&&run.skipped===0&&run.cancelled===0&&run.todo===0&&named(control,run)}
// Decode only the scalar styles emitted by Node TAP; unknown syntax fails closed.
function errorMessage(text){
 const lines=text.split('\n'),i=lines.findIndex(x=>/^  error: /.test(x));if(i<0)return null
 const scalar=lines[i].slice('  error: '.length)
 if(['|-','|','|+'].includes(scalar)){
  const body=[];for(let k=i+1;k<lines.length;k++){if(lines[k].startsWith('    '))body.push(lines[k].slice(4));else if(lines[k]==='')body.push('');else break}
  return body.join('\n')
 }
 if(scalar.startsWith('"')){try{const value=JSON.parse(scalar);return typeof value==='string'?value:null}catch{return null}}
 if(scalar.startsWith("'")){if(!scalar.endsWith("'"))return null;return scalar.slice(1,-1).replaceAll("''", "'")}
 if(!scalar||/^[>|&*!{[]/.test(scalar))return null
 return scalar
}
function intended(control,run,phase){
 const oracle=control.oracles[phase],stackToken=control.test.file+':'+oracle.startLine+':'
 const stack=run.text.match(/\n  operator: [^\n]*\n  stack:\s*\|-?\n((?: {4,}[^\n]*\n?)*)/)
 const firstFrame=stack?.[1].split('\n').find(line=>/:\d+:\d+\)?\s*$/.test(line))||''
 const operator=run.text.match(/^\s{2}operator:\s*['"]([^'"]+)['"]\s*$/m)?.[1]||''
 const message=errorMessage(run.text)
 const codes=[...run.text.matchAll(/^\s{2}code:\s*['"]([^'"]+)['"]\s*$/gm)].map(m=>m[1])
 const messageMatches=typeof message==='string'&&control.errorMessageIncludes.length>=2&&control.errorMessageIncludes.every(t=>message.includes(t))
 return {accepted:run.status===1&&run.signal===null&&!run.timedOut&&!run.spawnError&&run.tests===1&&run.pass===0&&run.fail===1&&run.skipped===0&&run.cancelled===0&&run.todo===0&&codes.length===1&&codes[0]==='ERR_ASSERTION'&&named(control,run)&&operator===control.expectedOperator&&messageMatches&&firstFrame.includes(stackToken),operator,message,messageMatches,firstFrame,stackToken,expected:oracle}
}
function ownedDir(dir,kind){if(!dir||!new RegExp('^'+g.BASE.replace(/[.*+?^${}()|[\]\\]/g,'\\$&')+'-'+kind+'-[A-Za-z0-9]+$').test(dir)||path.dirname(fs.realpathSync(dir))!==fs.realpathSync(path.dirname(g.BASE)))g.fail('Exact owned receipt directory required')}
function completion(ctx,pinSHA,dir,hash){if(!/^[a-f0-9]{64}$/.test(hash||''))g.fail('Exact completion SHA required');const raw=fs.readFileSync(path.join(dir,'complete.json'));if(g.sha(raw)!==hash)g.fail('Completion drift');const c=JSON.parse(raw);if(c.pinsSHA!==pinSHA||c.root!==g.ROOT||c.head!==ctx.pins.head||c.branch!==ctx.pins.branch)g.fail('Completion identity');const r=fs.readFileSync(path.join(dir,'receipt.json'));if(g.sha(r)!==c.receiptSHA256)g.fail('Receipt drift');if(Object.keys(c.allFileHashes||{}).length!==10)g.fail('All ten files required');for(const[f,h]of Object.entries(c.allFileHashes))if(h!==g.sha(ctx.snapshots[f]?.[c.phase]?.bytes||''))g.fail('Whole source/test phase mismatch');return{c,records:JSON.parse(r)}}
function reload(dir,record,label,control){const p=path.join(dir,control.id+'-'+label+'.tap'),commandPath=path.join(dir,control.id+'-'+label+'.command.json');if(record.log!==p||g.sha(fs.readFileSync(p))!==record.logSHA256)g.fail('TAP path/hash drift');const raw=fs.readFileSync(commandPath);if(g.sha(raw)!==record.commandSHA256)g.fail('Command hash drift');const cmd=JSON.parse(raw);if(cmd.executable!==process.execPath||cmd.cwd!==g.ROOT||JSON.stringify(cmd.args)!==JSON.stringify(control.command.slice(1))||cmd.timeoutMs!==(control.timeoutMs||45000)||cmd.detached!==true)g.fail('Command/timeout differs');const text=fs.readFileSync(p,'utf8');return{...record,...parseTAP(text),text}}
function validateControlsReceipt(ctx,pinSHA,dir,hash,phase){ownedDir(dir,'mutations-'+phase);const{c,records}=completion(ctx,pinSHA,dir,hash);if(c.phase!==phase||c.verified!==14||c.cleanRuns!==28||records.length!==14||JSON.stringify(c.controlIds)!==JSON.stringify(ctx.controls.map(c=>c.id)))g.fail('All14 ordered controls required');for(let i=0;i<14;i++){const control=g.activate(ctx.controls[i],phase),r=records[i];if(r.id!==control.id||r.phase!==phase||r.verified!==true||r.sourceBefore!==control.sourceSha256||r.sourceFault!==control.mutatedSha256||r.actualSourceFault!==control.mutatedSha256||r.sourceRestored!==control.sourceSha256||r.byteExactRestored!==true)g.fail('Phase-aware original/mutant/restore mismatch');for(const p of ctx.pins.readset)if(r.readsetBefore?.[p.file]!==p.sha||r.readsetAfter?.[p.file]!==p.sha)g.fail('Readset proof missing');if(!clean(control,reload(dir,r.before,'before',control))||!intended(control,reload(dir,r.fault,'fault',control),phase).accepted||!clean(control,reload(dir,r.after,'after',control)))g.fail('Clean/red/restore oracle failed: '+control.id)}return{dir,completionSHA:hash,phase,verified:14,logFiles:42}}
function cohortCommand(ctx){return['node','--import','./tests/server-only-register.cjs','--import','tsx','--test','--test-concurrency=1',...ctx.evidence.files.filter(f=>f.startsWith('tests/'))]}
function cohortClean(ctx,run,phase){const n=ctx.counts[phase],actual=[...run.text.matchAll(/^# Subtest: (.+)$/gm)].map(m=>m[1]).sort(),expected=Object.entries(ctx.snapshots).flatMap(([f,s])=>f.startsWith('tests/')?[...s[phase].rows.keys()]:[]).sort();return run.status===0&&run.signal===null&&!run.timedOut&&!run.spawnError&&run.tests===n&&run.pass===n&&run.fail===0&&run.skipped===0&&run.cancelled===0&&run.todo===0&&JSON.stringify(actual)===JSON.stringify(expected)}
function validateCohortReceipt(ctx,pinSHA,dir,hash,phase){ownedDir(dir,'cohort-'+phase);const{c,records}=completion(ctx,pinSHA,dir,hash);if(c.phase!==phase||c.verified!==ctx.counts[phase]||records.length!==1)g.fail('Complete bounded cohort proof required');const control={id:'cohort',command:cohortCommand(ctx),timeoutMs:180000},run=reload(dir,records[0],'full',control);if(!cohortClean(ctx,run,phase))g.fail('Full cohort count/selection/clean mismatch');return{dir,completionSHA:hash,phase,verified:ctx.counts[phase]}}
function validateTransferReceipt(ctx,pinSHA,cohortDir,cohortSHA,controlsDir,controlsSHA){return{cohort:validateCohortReceipt(ctx,pinSHA,cohortDir,cohortSHA,'transfer'),controls:validateControlsReceipt(ctx,pinSHA,controlsDir,controlsSHA,'transfer')}}
module.exports={parseTAP,named,clean,errorMessage,intended,validateTransferReceipt,validateControlsReceipt,cohortCommand,cohortClean,validateCohortReceipt}
