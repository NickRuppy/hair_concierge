// Main's read-only full-cohort receipt checker. Does not run native or import tests.
const fs=require('node:fs'),path=require('node:path'),g=require('./operator-guard.cjs'),proof=require('./operator-proof.cjs');
const PIN_SHA='4ac6d1b4344d4296e8c91668d9c62bc352869ce8636f4d2c54f42612bf118a3d';
if(process.argv.length!==5)g.fail('Use <before|transfer|cut> <closed-native.tap> <command.json>');

console.log(JSON.stringify({status:'PASS_RECORDED_FULL_NATIVE_TAP',phase,totals,logPath:path.resolve(logPath),logSHA256:g.sha(raw),commandSHA256:g.sha(commandBytes),pinsSHA:PIN_SHA,counts:ctx.counts,limits:'Main must attest closed process exit 0; text validation cannot prove process status.
