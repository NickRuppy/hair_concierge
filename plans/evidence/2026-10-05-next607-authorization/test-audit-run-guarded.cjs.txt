const fs=require('node:fs');
const cp=require('node:child_process');
const path=require('node:path');
const label=process.argv[2];
const root='/Users/nick/AI_work/hair_conscierge/.worktrees/test-audit-pruning';
const files=JSON.parse(fs.readFileSync('/tmp/test-audit-node-files.json','utf8')).filter(p=>fs.existsSync(path.join(root,p)));
const output='/tmp/test-audit-'+label;
if(fs.existsSync(output))throw Error('refuse existing evidence '+output);
fs.mkdirSync(output);
fs.writeFileSync(output+'/files.json',JSON.stringify(files,null,2));
const args=['/tmp/hc-test-audit-tools/node_modules/c8/bin/c8.js','--merge-async','--all','--src','src','--src','scripts','--src','apps','--src','packages','--include','src/**/*.{ts,tsx,js,mjs,cjs}','--include','scripts/**/*.{ts,tsx,js,mjs,cjs}','--include','apps/**/*.{ts,tsx,js,mjs,cjs}','--include','packages/**/*.{ts,tsx,js,mjs,cjs}','--exclude','**/*.d.ts','--exclude','**/*.test.*','--exclude','**/*.spec.*','--exclude','**/.next/**','--reporter=json-summary','--reporter=json','--reporter=lcovonly','--reporter=text-summary','--reports-dir='+output,process.execPath,'--enable-source-maps','--import','./tests/server-only-register.cjs','--import','tsx','--test','--test-concurrency=4','--test-reporter=tap','--test-reporter-destination='+output+'/tests.tap',...files];
fs.writeFileSync(output+'/command.json',JSON.stringify({node:process.version,root,args},null,2));
console.log('Running',files.length,'files; evidence',output);
const logfile=fs.openSync(output+'/run.log','w');
const started=Date.now();
const result=cp.spawnSync(process.execPath,args,{cwd:root,stdio:['ignore',logfile,logfile],env:process.env});
fs.closeSync(logfile);
fs.writeFileSync(output+'/result.json',JSON.stringify({status:result.status,signal:result.signal,error:result.error?.message,elapsedMs:Date.now()-started},null,2));
console.log('Complete:',result.status,'elapsed seconds:',(Date.now()-started)/1000);
process.exitCode=result.status??1;
