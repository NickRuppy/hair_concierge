const fail=x=>{throw Error(x)},read=p=>fs.readFileSync(p,'utf8');
function parse(file,text){let sf=ts.createSourceFile(file,text,ts.ScriptTarget.Latest,true,file.endsWith('tsx')?ts.ScriptKind.TSX:ts.ScriptKind.TS);if(sf.parseDiagnostics.length)fail('parse '+file);return sf}
function collect(file,text){let sf=parse(file,text),a=[],s=[];function walk(n){if(ts.isCallExpression(n)){let name=n.expression.getText(sf);if(name==='test'){let cb=n.arguments.at(-1);if(ts.isArrowFunction(cb)||ts.isFunctionExpression(cb)){let end=n.end;if(text[end]===';')end++;s.push({title:n.arguments[0].getText(sf),start:n.getStart(sf),end,body:text.slice(n.getStart(sf),end)})}}if(name.startsWith('assert.'))a.push({name,args:n.arguments.map(x=>x.getText(sf))})}ts.forEachChild(n,walk)}walk(sf);return {s,a}}
function tokens(x){const scan=ts.createScanner(ts.ScriptTarget.Latest,true,ts.LanguageVariant.Standard,x);let out=[],t;while((t=scan.scan())!==ts.SyntaxKind.EndOfFileToken)out.push(scan.getTokenText());return out.join('\u0000')}
function key(a){let n=a.name==='assert.ok'?1:2;return a.name+'('+a.args.slice(0,n).map(x=>tokens(x.replace(/textSequence\(real\)/g,'textSequence(realEmptyScreen())'))).join(',')+')'}
