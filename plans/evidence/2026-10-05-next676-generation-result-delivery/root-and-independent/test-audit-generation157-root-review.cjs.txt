const fs = require('node:fs');
const assert = require('node:assert/strict');
const crypto = require('node:crypto');
const kit = '/tmp/test-audit-generation157-tools-v2-orjded8b';
const root = '/Users/nick/AI_work/hair_conscierge/.worktrees/test-audit-pruning';
const proof = require(kit + '/operator-proof.cjs');
const sha = b => crypto.createHash('sha256').update(b).digest('hex');
const controls = JSON.parse(fs.readFileSync(kit + '/operator-controls-normalized.json'));
const dirs = process.argv.slice(2);
assert.ok(dirs.length === 1 || dirs.length === 2);
const all = [];
for (const [index, dir] of dirs.entries()) {
  const phase = index ? 'cut' : 'transfer';
  const receiptBytes = fs.readFileSync(dir + '/receipt.json');
  const records = JSON.parse(receiptBytes);
  const complete = JSON.parse(fs.readFileSync(dir + '/complete.json'));
  assert.equal(complete.receiptSHA256, sha(receiptBytes));
  assert.equal(records.length, 18);
  assert.equal(complete.verified, 18);
  assert.equal(complete.cleanRuns, 36);
  assert.equal(complete.phase, phase);
  assert.equal(complete.pinsSHA, 'b41d3a0d1a4021f978e0458b7e7a6c21881541ff195bb662c28e0d59ea917d70');
  const reviews = records.map((r, i) => {
    const c = controls[i];
    assert.equal(r.id, c.id);
    assert.equal(r.verified, true);
    assert.equal(r.byteExactRestored, true);
    assert.equal(r.sourceBefore, c.sourceSha256);
    assert.equal(r.sourceRestored, c.sourceSha256);
    assert.equal(r.actualSourceFault, c.mutatedSha256);
    assert.equal(sha(fs.readFileSync(root + '/' + c.file)), c.sourceSha256);
    const runs = {};
    for (const label of ['before', 'fault', 'after']) {
      const bytes = fs.readFileSync(dir + '/' + c.id + '-' + label + '.tap');
      assert.equal(sha(bytes), r[label].logSHA256);
      const commandBytes = fs.readFileSync(dir + '/' + c.id + '-' + label + '.command.json');
      assert.equal(sha(commandBytes), r[label].commandSHA256);
      const command = JSON.parse(commandBytes);
      assert.equal(command.cwd, root);
      assert.equal(c.command[0], 'node');
      assert.equal(command.executable, '/usr/local/bin/node');
      assert.deepEqual(command.args, c.command.slice(1));
      runs[label] = {...r[label], ...proof.parseTAP(bytes.toString()), text:bytes.toString()};
    }
    assert.ok(proof.clean(c, runs.before));
    assert.ok(proof.clean(c, runs.after));
    const parsed = proof.intended(c, runs.fault, phase);
    assert.ok(parsed.accepted, c.id);
    if (index) {
      assert.equal(parsed.operator, all[0].reviews[i].operator);
      assert.equal(parsed.message, all[0].reviews[i].message);
      assert.deepEqual(parsed.typedValues, all[0].reviews[i].typedValues);
    }
    return {id:c.id, operator:parsed.operator, firstFrame:parsed.firstFrame,
      message:parsed.message, typedValues:parsed.typedValues,
      logSHA256:r.fault.logSHA256, byteExactRestored:true, cleanBefore:true, cleanAfter:true};
  });
  all.push({phase, dir, reviews, completionSHA256:sha(fs.readFileSync(dir + '/complete.json'))});
}
const output = {status:'PASS', requiredControls:18*dirs.length, cleanRuns:36*dirs.length,
  completeDecodedDiagnosticsIdenticalAcrossPhases:dirs.length === 2, all,
  readLimits:'Complete printed object projections and custom SSR messages/witnesses validated by the reviewed strict decoder. Root separately inspects scalar/schema diagnostics and every intended FIRST assertion/witness; repeated decorative HTML is not claimed as a full human read. Source evidence establishes undefined and container semantics omitted by TAP.'};
fs.writeFileSync('/tmp/test-audit-generation157-root-' + (dirs.length === 2 ? 'final' : 'transfer') + '-diagnostic-review.json', JSON.stringify(output,null,2)+'\n');
console.log(JSON.stringify({status:output.status, requiredControls:output.requiredControls, cleanRuns:output.cleanRuns, completions:all.map(x=>x.completionSHA256), readLimits:output.readLimits}));
