'use strict';
const fs=require('fs'),path=require('path'),crypto=require('crypto');
const base=__dirname,root='/Users/nick/AI_work/hair_conscierge/.worktrees/test-audit-pruning';
const ts=require(path.join(root,'node_modules/typescript'));
const json=f=>JSON.parse(fs.readFileSync(path.join(base,f),'utf8')),sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const m=json('manifest-draft.json'),c=json('candidates.json'),ledger=json('ledger.json');
function rows(f,s){const sf=ts.createSourceFile(f,s,ts.ScriptTarget.Latest,true,f.endsWith('.tsx')?ts.ScriptKind.TSX:ts.ScriptKind.TS);if(sf.parseDiagnostics.length)throw Error('parse '+f);const out=[];function walk(n){if(ts.isCallExpression(n)&&n.expression.getText(sf)==='test'){const cb=n.arguments.find(x=>ts.isArrowFunction(x)||ts.isFunctionExpression(x));if(cb)out.push({name:ts.isStringLiteral(n.arguments[0])?n.arguments[0].text:n.arguments[0].getText(sf),callback:cb.getText(sf)});}ts.forEachChild(n,walk)}walk(sf);return out;}
const totals={},sets={};for(const p of ['before','transfer','cut']){sets[p]=new Map;let total=0;for(const r of m.phases[p]){const s=fs.readFileSync(path.join(base,p,r.file),'utf8');if(sha(s)!==r.sha256)throw Error('phase pin '+p+' '+r.file);const rr=rows(r.file,s);if(rr.length!==r.sites)throw Error('count');for(const x of rr)sets[p].set(r.file+'\0'+x.name,x.callback);total+=rr.length;}totals[p]=total;}if(JSON.stringify(totals)!==JSON.stringify({before:157,transfer:157,cut:150}))throw Error('total');
const donors=new Set(c.map(x=>x.donor.file+'\0'+x.donor.name)),keepers=new Set(c.filter(x=>x.classification==='C').flatMap(x=>x.keepers.map(k=>k.file+'\0'+k.name)));let untouched=0,heldF=0;
for(const [key,b]of sets.before){const t=sets.transfer.get(key),a=sets.cut.get(key);if(!t)throw Error('missing transfer');if(donors.has(key)){if(b!==t||a!==undefined)throw Error('donor change');continue;}if(t!==a)throw Error('cut survivor changed');if(!keepers.has(key)){if(b!==t)throw Error('unrelated changed');untouched++;}}
for(const r of ledger.cases.filter(x=>x.verdict==='F')){const key=r.file+'\0'+r.name;if(sets.before.get(key)!==sets.cut.get(key))throw Error('held F changed');heldF++;}
for(const cc of c.filter(x=>x.classification==='C'))for(const k of cc.keepers){const b=k.before.callback,t=k.transfer.callback;const sf=ts.createSourceFile('callback.ts','const keeper='+b,ts.ScriptTarget.Latest,true);function walk(n){if(ts.isExpressionStatement(n)&&/^assert\./.test(n.getText(sf))&&!t.includes(n.getText(sf)))throw Error('original assertion lost '+k.name);ts.forEachChild(n,walk)}walk(sf);}
for(const r of json('readsets.json'))if(sha(fs.readFileSync(path.join(root,r.file)))!==r.sha256)throw Error('readset drift '+r.file);
for(const r of m.phases.before)if(sha(fs.readFileSync(path.join(root,r.file)))!==r.sha256)throw Error('original test drift');
for(const x of json('controls.json')){const source=fs.readFileSync(path.join(root,x.owner),'utf8');if(sha(source)!==x.sourceSHA256||source.split(x.uniqueFrom).length!==2)throw Error('control original '+x.id);const mutant=source.replace(x.uniqueFrom,x.to),saved=fs.readFileSync(path.join(base,x.mutantSnapshot),'utf8');if(saved!==mutant||sha(saved)!==x.mutantSHA256)throw Error('mutant mismatch');rows(x.owner,saved);}
console.log(JSON.stringify({staticOnly:true,repositoryWrites:0,ownersImported:0,testsExecuted:0,childrenStarted:0,counts:totals,unrelatedCallbacks:untouched,heldF,controls:json('controls.json').length,allProspectiveTSParse:true,currentReadsetsMatch:true}));
