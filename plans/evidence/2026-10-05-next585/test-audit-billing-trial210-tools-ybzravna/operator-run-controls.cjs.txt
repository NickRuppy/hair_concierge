function killOwnedGroup(signal){if(activeChild?.pid){try{process.kill(-activeChild.pid,signal)}catch(error){if(error.code!=='ESRCH')throw error}}}

async function run(control,label){
  const log=path.join(dir,control.id+'-'+label+'.tap'),fd=fs.openSync(log,'wx'),args=control.command.slice(1)
  let timer,child,timedOut=false,result,spawnError
  try{

    timer=setTimeout(()=>{timedOut=true;killOwnedGroup('SIGKILL')},45000)
    result=await new Promise(resolve=>{child.once('error',error=>{spawnError=String(error);resolve({status:null,signal:null})});child.once('close',(status,signal)=>resolve({status,signal}))})
  }finally{
    clearTimeout(timer)
    // Only the group created by this detached child is eligible for cleanup, including
    // a descendant left after a failing resource-abort assertion. Never enumerate/kill peers.
    killOwnedGroup('SIGKILL');activeChild=null;fs.closeSync(fd)
  }
  const text=fs.readFileSync(log,'utf8')
  return{...result,...parseTAP(text),timedOut,spawnError:spawnError||null,log,text}
}

function named(control,run){return run.text.includes('# Subtest: '+control.test.name+'\n')}
function clean(control,run){return run.status===0&&run.signal===null&&!run.timedOut&&!run.spawnError&&run.tests===1&&run.pass===1&&run.fail===0&&run.skipped===0&&run.cancelled===0&&run.todo===0&&named(control,run)}
function intended(control,run){

 const operator=run.text.match(/^\s{2}operator:\s*['"]([^'"]+)['"]\s*$/m)?.[1]||''
 const message=run.text.match(/\n  error: ([\s\S]*?)(?=\n  (?:code|name|expected|actual|operator|stack):)/)?.[1]||''
 const diagnostics=control.errorMessageIncludes.every(token=>message.includes(token))
 const codes=[...run.text.matchAll(/^\s{2}code:\s*['"]([^'"]+)['"]\s*$/gm)].map(m=>m[1])
 return{accepted:run.status===1&&run.signal===null&&!run.timedOut&&!run.spawnError&&run.tests===1&&run.pass===0&&run.fail===1&&run.skipped===0&&run.cancelled===0&&run.todo===0&&codes.length===1&&codes[0]==='ERR_ASSERTION'&&named(control,run)&&operator===control.expectedOperator&&diagnostics&&frame.includes(stackToken),expected:oracle,expectedOperator:control.expectedOperator,operator,errorMessage:message,errorMessageMatches:diagnostics,firstAssertionFrame:frame,stackToken,named:named(control,run)}
}
