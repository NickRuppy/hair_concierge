279: export default async function RoutinePage() {
280:   // T12 (freemium-scanner-first PR3): the free tier gets the framed „Beispiel" Routine
281:   // instead of its own (empty, stage-gated) one — the only branch this page gained;
282:   // premium and flag-off fall straight through to today's behaviour below.
283:   //
284:   // Pre-boundary fix wave (F2): the tier check and `resolveDefaultRoutinePage` now start
285:   // CONCURRENTLY instead of the tier gating serially ahead of the resolver — the previous
286:   // order added one `auth.getUser()` plus a billing/moderator composite in front of every
287:   // PREMIUM render, which is the common case. The trade is a resolver read a FREE render
288:   // now performs and discards (never a write — `resolveRoutinePage` is read-only and
289:   // catches its own errors) for zero added latency on the premium path. Semantics are
290:   // unchanged: a free render still never reaches `resolved`'s output, and
291:   // `resolveGatedPageMode` itself still fails closed to premium on flag-off or an
292:   // entitlement-source outage.
293:   //
294:   // T17 widens that one branch from two states to three: `"example"` is the never-paid
295:   // free tier (unchanged), `"keepsake"` is a LAPSED owner who keeps reading their OWN
296:   // Routine with every mutation locked, and `"premium"` still falls straight through to
297:   // today's page. The keepsake read starts only after the mode is known — it is a second
298:   // owner read that a premium or free render must never pay for.
299:   const [pageMode, resolved] = await Promise.all([
300:     resolveGatedPageMode(),
301:     resolveDefaultRoutinePage(),
302:   ])
303:   if (pageMode === "keepsake") {
304:     const keepsake = await resolveKeepsakeRoutinePage(keepsakeDeps)
305:     if (keepsake.kind === "keepsake") {
306:       return (
307:         <PersonalPlanRoutineClient
308:           initialView={keepsake.view}
309:           enabled={false}
310:           portfolioPresentation={keepsake.portfolioPresentation}
311:           initialRefinementBanner={null}
312:           merklisteEnabled={isFreemiumScannerFirstEnabled()}
313:           keepsake
314:         />
315:       )
316:     }
317:     // PR5 review fix (Z2): a lapsed owner with no accepted Routine keeps the keepsakes they
318:     // DO have — their Merkliste is right here, read-only — and sees an honest empty Routine
319:     // instead of the „Beispiel" page.
320:     if (keepsake.kind === "no_routine") {
321:       return <KeepsakeNoRoutineState merklisteEnabled={isFreemiumScannerFirstEnabled()} />
322:     }
323:     // The keepsake read itself failed (`unavailable`): fall back to the free tier's own
324:     // page rather than inventing a third state on an untrusted signal.
325:     return <GatedRoutineExample />
326:   }
327:   const renderGatedExample = pageMode === "example"
328:   // F3: verified with two production builds + a live network capture — the suspected
329:   // bundle bloat did not reproduce. `GatedPreview`'s (and therefore `PremiumSheet`'s)
330:   // client chunk set is IDENTICAL to `RoutinePageClient`'s own (see the matching comment
331:   // in `app/chat/page.tsx`), i.e. that code already ships on this route via a chunk shared
332:   // for unrelated reasons, so a dynamic import here would move nothing. Kept static.
333:   if (renderGatedExample) return <GatedRoutineExample />
334: 
335:   if (resolved.kind === "legacy") {
336:     // T16: the „Gemerkt" section (Merkliste's new home, replacing the scan flow's in-sheet
337:     // list) is gated on the freemium flag itself, independent of tier — the section's OWN
338:     // fetch (`/api/scan/wishlist`) is what tells premium from free (403 for free, hidden
339:     // silently), but the flag has to gate whether `RoutinePageClient` attempts that fetch
340:     // AT ALL. Without this, an existing paid subscriber with the flag OFF would see a
341:     // brand-new section today's Routine page never had — flag-off must stay byte-identical.
342:     return <RoutinePageClient merklisteEnabled={isFreemiumScannerFirstEnabled()} />
343:   }
344: 
345:   if (resolved.kind === "unavailable") {
346:     return <RoutineUnavailableState />
347:   }
348: 
349:   return (
350:     <PersonalPlanRoutineClient
351:       initialView={resolved.view}
352:       enabled={resolved.enabled}
353:       portfolioPresentation={resolved.portfolioPresentation}
354:       initialRefinementBanner={resolved.refinementBanner}
355:       // Fix round 1 (F1): every freemium-provisioned buyer and every current subscriber
356:       // resolves HERE, not to the `legacy` branch above — the same flag gate has to reach
357:       // this branch too, or the „Gemerkt" section (and the scanner bookmark's
358:       // `/routine#gemerkt` deep-link) is unreachable for the exact cohort it exists for.
359:       merklisteEnabled={isFreemiumScannerFirstEnabled()}
360:     />
361:   )
362: }

153:       planId: source.personalPlanId,
154:       routineVersionId: activeVersion.id,
155:       refinedVersionId: activeVersion.payload.source.refinedVersionId,
156:     }
157:     const [accepted, profile] = await Promise.all([
158:       deps.adaptRoutine({ client, activeVersion, contractVersion }),
159:       deps.loadProfile({
160:         client,
161:         userId,
162:         planId: source.personalPlanId,
163:         refinedVersionId: activeVersion.payload.source.refinedVersionId,
164:       }),
165:     ])
166:     const familyPayloads = protocols.map((protocol) =>
167:       applicationFamilyTemplateV2Schema.parse(protocol.payload),
168:     )
169:     const applicationInput = {
170:       routineItems: accepted.routineItems,
171:       unresolvedRoutineItems: accepted.unresolvedRoutineItems,
172:       profile,
173:       dayTypes: dayDefinitions.map((day) => ({ key: day.key, sortOrder: day.sortOrder })),
174:     }
175:     if (!accepted.applicationPointersV2) {
176:       throw new Error("application_v2_product_pointers_unavailable")
177:     }
178:     const compiled = compileApplicationViewV2({
179:       input: applicationInput,
180:       familyTemplates: familyPayloads,
181:       productPointers: accepted.applicationPointersV2,
182:     })
183:     for (const degraded of accepted.degradedItems) {
184:       deps.reportFailure({
185:         ...failureContext,
186:         durationMs: Date.now() - startedAt,
187:         reason: "product_guidance_unresolved",
188:         productId: degraded.productId,
189:         issueCode: degraded.issue,
190:       })
191:     }
192:     const pointerIssues = compiled.pointerIssues
193:     if (pointerIssues.length > 0) {
194:       for (const issue of pointerIssues) {
195:         deps.reportFailure({
196:           ...failureContext,
197:           durationMs: Date.now() - startedAt,
198:           reason: "product_guidance_unresolved",
199:           productId: issue.productId,
200:           issueCode: issue.reason,
201:         })
202:       }
203:     }
204:     const selectedFailure = selectedDayType
205:       ? compiled.failures.find((failure) => failure.dayType === selectedDayType)
206:       : undefined
207:     if (selectedFailure) {
208:       deps.reportFailure({
209:         ...failureContext,
210:         durationMs: Date.now() - startedAt,
211:         reason:
212:           selectedFailure.reason === "incomplete_guidance"
213:             ? "incomplete_guidance"
214:             : "missing_protocol",
215:       })
216:     }
217:     const view = toApplicationPageView({
218:       compiled,
219:       dayDefinitions,
220:       cadenceByDay: projectApplicationCadenceByDay({
221:         routineItems: accepted.routineItems,
222:         compiledDayKeys: compiled.days.map((day) => day.key),
223:       }),
224:     })
225:     if (selectedDayType) {
226:       if (view.state === "ready" && view.days.some((day) => day.dayType === selectedDayType)) {
227:         return { ...view, selectedDayType }
228:       }
229:       if (view.state === "no_complete_day" && view.restDay.dayType === selectedDayType) {
230:         return { state: "ready", days: [view.restDay], selectedDayType }
231:       }
232:       return { state: "day_unavailable", overviewHref: "/anwendung" }
233:     }
234:     return view
235:   } catch (error) {
236:     deps.reportFailure({
237:       ...failureContext,
238:       durationMs: Date.now() - startedAt,
239:       reason: failureReason(error),
240:       failureCode: failureCode(error),
241:     })
242:     return { state: "unavailable" }
243:   }
244: }

285: export default async function AnwendungPage({
286:   selectedDayType,
287: }: { selectedDayType?: ApplicationDayTypeKey } = {}) {
288:   // T12 (freemium-scanner-first PR3): the free tier gets the framed „Beispiel" Anwendung
289:   // instead of its own (stage-gated, empty) one — for the overview and for every
290:   // `/anwendung/[dayType]` deep link, which re-renders this same component.
291:   //
292:   // Pre-boundary fix wave (F2): the tier check and `resolveDefaultAnwendungPage` now start
293:   // CONCURRENTLY instead of the tier gating serially ahead of the resolver — the previous
294:   // order added one `auth.getUser()` plus a billing/moderator composite in front of every
295:   // PREMIUM render, which is the common case. The trade is a resolver read a FREE render
296:   // now performs and discards (never a write — `resolveAnwendungPage` is read-only; its
297:   // `reportFailure` calls only fire on genuine compile failures, unrelated to the tier) for
298:   // zero added latency on the premium path. Semantics are unchanged: a free render still
299:   // never reaches `view`, and `resolveGatedPageMode` itself still fails closed to
300:   // premium on flag-off or an entitlement-source outage.
301:   //
302:   // T17 widens that one branch from two states to three: `"example"` is the never-paid
303:   // free tier (unchanged), `"keepsake"` is a LAPSED owner reading their OWN Anwendung
304:   // (same page, same compiler — sourced from their accepted Routine instead of live
305:   // journey entitlement; the page itself has no mutations to lock), and `"premium"` falls
306:   // straight through to today's page.
307:   const [pageMode, { view, durationMs }] = await Promise.all([
308:     resolveGatedPageMode(),
309:     resolveDefaultAnwendungPage(selectedDayType),
310:   ])
311:   if (pageMode === "keepsake") {
312:     const keepsake = await resolveDefaultAnwendungPage(selectedDayType, { keepsake: true })
313:     // A keepsake read that produced nothing renderable falls back to the free tier's own
314:     // page rather than showing an owner a "feature disabled" dead end.
315:     if (keepsake.view.state !== "feature_disabled" && keepsake.view.state !== "unavailable") {
316:       return <RouteAwareApplicationPage view={keepsake.view} />
317:     }
318:     return <GatedAnwendungExample />
319:   }
320:   const renderGatedExample = pageMode === "example"
321:   // F3: verified with two production builds + a live network capture — the suspected
322:   // bundle bloat did not reproduce. `GatedPreview`'s (and therefore `PremiumSheet`'s)
323:   // client chunk set is IDENTICAL to `ApplicationPage`'s own (see the matching comment in
324:   // `app/chat/page.tsx`), i.e. that code already ships on this route via a chunk shared
325:   // for unrelated reasons, so a dynamic import here would move nothing. Kept static.
326:   if (renderGatedExample) return <GatedAnwendungExample />
327: 
328:   const internalComputeMs =
329:     process.env.PERSONAL_PLAN_APPLICATION_PERFORMANCE_MARKER_ENABLED === "true"
330:       ? Math.round(durationMs * 100) / 100
331:       : undefined
332:   return <RouteAwareApplicationPage view={view} internalComputeMs={internalComputeMs} />
333: }
