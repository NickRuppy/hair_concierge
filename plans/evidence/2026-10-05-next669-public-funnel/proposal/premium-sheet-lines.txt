53: /**
54:  * The Premium sheet (T13, freemium-scanner-first PR4) — the program's one conversion
55:  * surface. It replaces T5's stub in place: same opener contract (`open` / `context` /
56:  * `onClose`), so every gate that already opens it — the scan verdict (T9), the trigger
57:  * cards (T10), the gated „Beispiel" pages (T12) — is untouched.
58:  *
59:  * Three things, in this order:
60:  *  1. what the user just bumped into, explained — `orderedBenefits` puts that feature
61:  *     first (plum, the repo's selected/accent colour) and fills the other two slots from
62:  *     the core order, so the sheet never reads as a single-feature paywall;
63:  *  2. the three plans at standard-catalog prices (`@/lib/premium-sheet/pricing` — never
64:  *     the launch catalog, whatever the launch-pricing flag says), Vierteljährlich
65:  *     preselected and marked „Beliebteste Wahl" (docket rework R2, ruling A3);
66:  *  3. one coral CTA, and an escape that is always one tap away. Declining costs nothing:
67:  *     the free scanner keeps working exactly as before.
68:  *
69:  * **T14 — contextual purchase completion.** The CTA no longer dismisses: it swaps the
70:  * sheet's body for Stripe embedded checkout plus the offer page's native PayPal button
71:  * (docket rework R1), in place. The whole purchase happens inside this component, which is
72:  * why the opener contract still does not change — no navigation, no `/welcome`, no lost
73:  * scan state, whichever provider the buyer picks.
74:  *
75:  * What this component may and may not conclude:
76:  *  - Stripe's `onComplete` moves it to „prüfen", never to unlocked. Only
77:  *    `POST /api/freemium/purchase/complete` — which re-reads the Session from Stripe —
78:  *    unlocks anything. The state machine (`lib/premium-sheet/purchase-state.ts`) makes that
79:  *    the only reachable path.
80:  *  - On unlock it refreshes the router, tells the opener (`onUnlocked`) and closes. The
81:  *    gates are server-rendered from the entitlement tier, so the refresh is what turns the
82:  *    „Beispiel" frame into the buyer's real content — but a surface that holds its own
83:  *    client-side tier state (the scanner's Merken bookmark) cannot learn from a refresh,
84:  *    which is why the callback exists (fix round 1, F1).
85:  *  - Every failure lands back on the plan rows inside the same open sheet. Nothing is
86:  *    navigated and nothing about the free session is touched.
87:  */

154:   /**

156:    * asked whether money actually moved.
157:    *
158:    * `silentOnError` is what makes polling safe: a network blip on poll 3 must not demote a

161:    *

163:    * resumed purchase is not a fresh, user-initiated attempt, so a transport/5xx error here

171:    */

173:     async (
174:       sessionId: string,
175:       options: { silentOnError?: boolean; treatErrorAsPending?: boolean } = {},

178:         status?: unknown
179:         routineReady?: unknown
180:         retryable?: unknown
181:         reason?: unknown

187:           method: "POST",
188:           headers: { "Content-Type": "application/json" },
189:           // Stripe or PayPal, decided by the reference itself (docket rework R1) — the only
190:           // place the two lanes differ on this side of the purchase.
191:           body: JSON.stringify(premiumSheetCompletionBody(sessionId)),
192:         })
193:         if (response.status === 429) {
194:           rateLimited = true
195:         } else {
196:           payload = (await response.json().catch(() => null)) as CompletionPayload
197:           if (!response.ok) payload = null
198:         }
199:       } catch {
200:         payload = null
201:       }
202:       if (payload?.status === "complete") {
203:         dispatchPurchase({
204:           type: "verification_complete",
205:           sessionId,
206:           routineReady: payload.routineReady === true,
207:         })
208:         return "settled"
209:       }
210:       if (payload?.status === "provisioning") {
211:         // Paid and entitled, plan not built. `retryable: false` stops the poll.
212:         dispatchPurchase({
213:           type: "verification_provisioning",
214:           sessionId,
215:           retryable: payload.retryable !== false,
216:         })
217:         return "settled"
218:       }
219:       if (payload?.status === "pending") {
220:         dispatchPurchase({ type: "verification_pending", sessionId })
221:         return "settled"
222:       }
223:       if (payload?.status === "failed") {
224:         dispatchPurchase({
225:           type: "verification_failed",
226:           sessionId,
227:           reason: verificationFailureReason(payload.reason),
228:         })
229:         return "settled"
230:       }
231:       if (rateLimited) {
232:         // Never an authoritative verdict, from any phase this is called from (a fresh
233:         // attempt, the resume lane, a poll, or a manual recheck) — always parks the purchase
234:         // at `pending` so the poll or the next manual tap can retry it, instead of leaving a
235:         // first-attempt call stuck in `verifying` with no dispatch at all (Codex fix wave
236:         // round 2, R3).
237:         dispatchPurchase({ type: "verification_pending", sessionId })
238:         return "rate_limited"
239:       }
240:       if (options.treatErrorAsPending) {
241:         dispatchPurchase({ type: "verification_pending", sessionId })
242:         return "settled"
243:       }
244:       if (options.silentOnError) return "settled"
245:       dispatchPurchase({ type: "verification_failed", sessionId, reason: "verification_failed" })
246:       return "settled"

254:   const verifyingSessionId = purchase.phase === "verifying" ? purchase.sessionId : null

432:    */

434:   // Once, ever. Openers pass inline callbacks, so every re-render changes this effect's

439:     if (unlockedRoutineReady === null || unlockHandledRef.current) return
440:     unlockHandledRef.current = true

444:       title: PREMIUM_SHEET_PURCHASE_COPY.unlockToast,

459:       type: "checkout_requested",

714: /**
715:  * The server's failure reason, narrowed to something the sheet can say in German (Codex fix
716:  * wave, Y4). Only the two lifecycle outcomes get their own line — a Session the buyer
717:  * abandoned and one that expired are different sentences, and both are different from „Wir
718:  * konnten deine Zahlung nicht bestätigen.", which is what every other reason falls back to.
719:  */
720: function verificationFailureReason(reason: unknown): PremiumSheetPurchaseFailure {
721:   if (reason === "checkout_session_expired") return "checkout_expired"
722:   if (reason === "checkout_session_abandoned") return "checkout_abandoned"
723:   // PayPal's equivalent of an expired Session: the checkout intent outlived its 24h TTL
724:   // (docket rework R1). Same sentence, because it is the same thing to the buyer.
725:   if (reason === "paypal_checkout_intent_expired") return "checkout_expired"
726:   // The duplicate guard already cancelled the second subscription and the buyer keeps the
727:   // access they had — a generic „nicht bestätigt" would misstate what happened (copy
728:   // polish wave): the payment was stopped on purpose, not left unverified.
729:   if (reason === "paypal_duplicate_checkout") return "subscription_already_active"
730:   return "verification_failed"
731: }
