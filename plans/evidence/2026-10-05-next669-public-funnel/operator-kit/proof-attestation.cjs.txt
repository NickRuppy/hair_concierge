function diagnostic(c){const d=c.diagnostic;return(d.kind==='omitted-undefined'?'':d.kind.startsWith('regex-')?'  expected:\n':field('expected',d.expected))+field('actual',d.kind==='regex-prefix'?d.syntheticShapeSample:d.actual)}

 for(const extra of[{timedOut:true},{signal:'SIGKILL'},{spawnError:'error'},{status:0}])ok(!proof.intended(c,run(text,extra),phase).accepted,'process rejection');
 for(const [from,to]of[["code: 'ERR_ASSERTION'","code: 'ERR_MODULE_NOT_FOUND'"],["name: 'AssertionError'","name: 'TypeError'"],["operator: '"+c.expectedOperator+"'","operator: '!='"],[c.test.file+':'+c.oracles[phase].startLine+':10',c.test.file+':1:10'],['# tests 1','# tests 0'],['# skipped 0','# skipped 1'],['# cancelled 0','# cancelled 1'],['# todo 0','# todo 1'],['# Subtest: '+c.test.name,'# Subtest: Wrong keeper'],['  error: |-','  error: >-']])ok(!accept(text.replace(from,to)),c.id+' rejects '+from);

 }

 }
 if(c.diagnostic.kind.startsWith('regex-')){
  ok(!accept(text.replace('  expected:\n','  expected: "{}"\n')),'regexp diagnostic requires typed empty object');
  ok(!accept(text.replace('  expected:\n','  expected:\n    nested: true\n')),'regexp nested expected rejected');
 }
 if(c.diagnostic.kind==='regex-prefix'){
  const sample=c.diagnostic.syntheticShapeSample,prefix=sample+'x'.repeat(10000-sample.length),actualLine='  actual: '+JSON.stringify(prefix)+'... 731 more characters\n';
  ok(accept(text.replace(/^  actual:.*\n/m,actualLine)),'supported inspect truncated prefix');
  ok(!accept(text.replace(/^  actual:.*\n/m,actualLine.replace('731 more','0 more'))),'nonpositive truncation rejected');
  ok(!accept(text.replace(/^  actual:.*\n/m,'  actual: '+JSON.stringify(sample)+'... 731 more characters\n')),'unsupported prefix length rejected');
  ok(!accept(text.replace(/^  actual:.*\n/m,'  actual: '+JSON.stringify('x'.repeat(10000))+'... 731 more characters\n')),'bad markup absent prefix rejected');
  ok(!accept(text.replace(c.diagnostic.fullHeader.replaceAll('\n','\n    '),'unrelated regex header')),'full regex header required');
  // A correct-looking nested expected cannot substitute for the required top-level RegExp object.
  ok(!accept(text.replace('  expected:\n','  auxiliary:\n    expected:\n')),'nested-only expected rejected');

 }

 ok(!accept(wrong),'nested correct frame cannot rescue wrong first outer frame');

}
