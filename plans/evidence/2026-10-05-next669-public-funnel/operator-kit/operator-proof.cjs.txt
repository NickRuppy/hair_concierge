'use strict'
// Shared read-only TAP proof validation. Cut revalidates logs instead of trusting a boolean.
const fs=require('node:fs'),path=require('node:path')
const g=require('./operator-guard.cjs')
function parseTAP(text){
 const number=key=>{const all=[...text.matchAll(new RegExp('^# '+key+' (\\d+)$','gm'))];return all.length===1?Number(all[0][1]):null}
 return {tests:number('tests'),pass:number('pass'),fail:number('fail'),skipped:number('skipped'),cancelled:number('cancelled'),todo:number('todo')}
}
function named(control,run){const names=[...run.text.matchAll(/^# Subtest: (.+)$/gm)].map(m=>m[1]);return names.length===1&&names[0]===control.test.name}
function clean(control,run){return run.status===0&&run.signal===null&&!run.timedOut&&!run.spawnError&&run.tests===1&&run.pass===1&&run.fail===0&&run.skipped===0&&run.cancelled===0&&run.todo===0&&named(control,run)}
// Decode only the scalar styles emitted by Node TAP; unknown syntax fails closed.
function errorMessage(text){
 const lines=text.split('\n'),i=lines.findIndex(x=>/^  error: /.test(x));if(i<0)return null
 const scalar=lines[i].slice('  error: '.length)
 if(['|-','|','|+'].includes(scalar)){
  const body=[];for(let k=i+1;k<lines.length;k++){if(lines[k].startsWith('    '))body.push(lines[k].slice(4));else if(lines[k]==='')body.push('');else break}
  return body.join('\n')
 }
 if(scalar.startsWith('"')){try{const value=JSON.parse(scalar);return typeof value==='string'?value:null}catch{return null}}
 if(scalar.startsWith("'")){if(!scalar.endsWith("'"))return null;return scalar.slice(1,-1).replaceAll("''", "'")}
 if(!scalar||/^[>|&*!{[]/.test(scalar))return null
 return scalar
}
function diagnosticValue(text,key){
 const lines=text.split('\n'),found=lines.map((v,i)=>v.startsWith('  '+key+':')?i:-1).filter(i=>i>=0);if(found.length!==1)return{ok:false}
 const i=found[0],tail=lines[i].slice(('  '+key+':').length).trimStart()
 function scalar(value){
   const quote=value[0]
   if(["'",'"','`'].includes(quote)){
     let out='',i=1,closed=false
     for(;i<value.length;i++){
       const char=value[i];if(char===quote){closed=true;i++;break}
       if(char!=='\\'){out+=char;continue}
       const e=value[++i],simple={'n':'\n','r':'\r','t':'\t','b':'\b','f':'\f','v':'\v','0':'\0','\\':'\\',"'":"'",'"':'"','`':'`'}
       if(Object.hasOwn(simple,e)){out+=simple[e];continue}
       if(e==='x'||e==='u'){const n=e==='x'?2:4,hex=value.slice(i+1,i+1+n);if(hex.length!==n||!/^[a-fA-F0-9]+$/.test(hex))return{ok:false};out+=String.fromCharCode(parseInt(hex,16));i+=n;continue}
       return{ok:false}
     }
     if(!closed)return{ok:false}
     const suffix=value.slice(i)
     if(!suffix)return{ok:true,value:out,truncated:false}
     const truncated=suffix.match(/^\.\.\. ([1-9]\d*) more characters$/)
     if(truncated&&out.length===10000)return{ok:true,value:out,truncated:true,remainingCharacters:Number(truncated[1])}
     return{ok:false}
   }
   if(value==='true'||value==='false'||value==='null')return{ok:true,value:JSON.parse(value)}
   if(/^-?(?:0|[1-9]\d*)(?:\.\d+)?$/.test(value))return{ok:true,value:Number(value)}
   if(value&&!/^[>|&*!{[]/.test(value))return{ok:true,value}
   return{ok:false}
 }
 if(['|-','|','|+'].includes(tail)){const body=[];for(let k=i+1;k<lines.length;k++){if(lines[k].startsWith('    '))body.push(lines[k].slice(4));else if(lines[k]==='')body.push('');else break}return{ok:true,value:body.join('\n')}}
 if(tail)return scalar(tail)
 const values=[];for(let k=i+1;k<lines.length&&lines[k].startsWith('    ');k++){const m=lines[k].match(/^    (\d+): (.*)$/);if(!m||Number(m[1])!==values.length)return{ok:false};const v=scalar(m[2]);if(!v.ok)return v;values.push(v.value)}
 return values.length?{ok:true,value:values}:{ok:true,value:{},emptyObject:true}
}
// This wrapper leaves the catalog decoder byte-exact. SSR proof concerns the visible prefix only.
function valueProof(control,text,message){
 const spec=control.diagnostic,actual=diagnosticValue(text,'actual'),expected=diagnosticValue(text,'expected');
 if(!spec||!actual.ok)return{accepted:false,reason:'missing/unsupported typed actual'};
 const exact=(a,b)=>typeof a===typeof b&&Object.is(a,b);
 if(spec.kind==='omitted-undefined'){
  const absent=!text.split('\n').some(line=>line.startsWith('  expected:'));
  const bound=control.id==='C1-04'&&control.expectedOperator==='strictEqual'&&control.test.file==='tests/gated-preview-component.test.tsx'&&control.test.name==='renders the frame, the Beispiel band, the benefit line and exactly one CTA'&&control.oracles.transfer.startLine===150&&control.oracles.cut.startLine===150&&control.oracles.transfer.text==='assert.equal(scroll.props["aria-label"], undefined)'&&control.oracles.cut.text===control.oracles.transfer.text;
  return{accepted:bound&&absent&&!actual.truncated&&exact(actual.value,'duplicate')&&message===spec.fullMessage,kind:spec.kind,expectedFieldAbsent:absent,actual};
 }
 if(!expected.ok)return{accepted:false,reason:'missing/unsupported typed expected'};
 if(spec.kind==='scalar')return{accepted:!actual.truncated&&!expected.truncated&&exact(actual.value,spec.actual)&&exact(expected.value,spec.expected)&&message===spec.fullMessage,kind:spec.kind,actual,expected};
 if(spec.kind==='regex-scalar')return{accepted:expected.emptyObject===true&&typeof actual.value==='string'&&!actual.truncated&&actual.value===spec.actual&&message===spec.fullMessage,kind:spec.kind,actual,expected};
 if(spec.kind==='regex-prefix'){
  const regex=new RegExp(spec.pattern,spec.flags),knownString=typeof actual.value==='string'&&(actual.truncated===undefined||typeof actual.truncated==='boolean');
  const header=typeof message==='string'&&message.startsWith(spec.fullHeader),badPrefix=knownString&&regex.test(actual.value)&&spec.requiredTokens.every(t=>actual.value.includes(t));
  return{accepted:expected.emptyObject===true&&header&&badPrefix,kind:spec.kind,actual,expected,fullHeader:header,prohibitedMarkupInPrintedPrefix:badPrefix,limits:'Typed printed string/prefix only. Explicit inspect truncation requires catalog length/shape checks. No assertion about an unseen HTML tail.'};
 }
 return{accepted:false,reason:'unknown diagnostic policy'};
}
function intended(control,run,phase){
 const oracle=control.oracles[phase],stackToken=control.test.file+':'+oracle.startLine+':'
 const stack=run.text.match(/\n  operator: [^\n]*\n  stack:\s*\|-?\n((?: {4,}[^\n]*\n?)*)/)
 const firstFrame=stack?.[1].split('\n').find(line=>/:\d+:\d+\)?\s*$/.test(line))||''
 const operator=run.text.match(/^\s{2}operator:\s*['"]([^'"]+)['"]\s*$/m)?.[1]||''
 const message=errorMessage(run.text)
 const diagnostic=valueProof(control,run.text,message)
 const codes=[...run.text.matchAll(/^\s{2}code:\s*['"]([^'"]+)['"]\s*$/gm)].map(m=>m[1])
 const names=[...run.text.matchAll(/^\s{2}name:\s*['"]([^'"]+)['"]\s*$/gm)].map(m=>m[1])
 const messageMatches=typeof message==='string'&&control.errorMessageIncludes.length>=2&&control.errorMessageIncludes.every(t=>message.includes(t))
 return {accepted:run.status===1&&run.signal===null&&!run.timedOut&&!run.spawnError&&run.tests===1&&run.pass===0&&run.fail===1&&run.skipped===0&&run.cancelled===0&&run.todo===0&&codes.length===1&&codes[0]==='ERR_ASSERTION'&&names.length===1&&names[0]==='AssertionError'&&named(control,run)&&operator===control.expectedOperator&&messageMatches&&firstFrame.includes(stackToken)&&diagnostic.accepted,operator,message,messageMatches,firstFrame,stackToken,expected:oracle,diagnostic}
}
function validateTransferReceipt(ctx,pinSHA,dir,completionSHA){
 if(!dir||!new RegExp('^'+g.BASE.replace(/[.*+?^${}()|[\]\\]/g,'\\$&')+'-mutations-transfer-[A-Za-z0-9]+$').test(dir)||path.dirname(fs.realpathSync(dir))!==fs.realpathSync(path.dirname(g.BASE)))g.fail('Exact owned transfer receipt directory required')
 if(!/^[a-f0-9]{64}$/.test(completionSHA||''))g.fail('Main must provide exact completion SHA256')
 const raw=fs.readFileSync(path.join(dir,'complete.json'));if(g.sha(raw)!==completionSHA)g.fail('Completion receipt hash mismatch')
 const complete=JSON.parse(raw),receiptBytes=fs.readFileSync(path.join(dir,'receipt.json'))
 if(complete.pinsSHA!==pinSHA||complete.root!==g.ROOT||complete.head!==ctx.pins.head||complete.branch!==ctx.pins.branch||complete.phase!=='transfer'||complete.verified!==20||complete.cleanRuns!==40||g.sha(receiptBytes)!==complete.receiptSHA256)g.fail('Incomplete/wrong transfer proof')
 if(JSON.stringify(complete.controlIds)!==JSON.stringify(ctx.controls.map(c=>c.id)))g.fail('Required20 controls absent or out of order')
 for(const[file,hash]of Object.entries(complete.allTestHashes||{})){if(!ctx.snapshots[file]||hash!==g.sha(ctx.snapshots[file].transfer.bytes))g.fail('Receipt test phase mismatch')}
 if(Object.keys(complete.allTestHashes||{}).length!==12)g.fail('Full cohort proof missing')
 const records=JSON.parse(receiptBytes);if(records.length!==20)g.fail('Twenty records required')
 function reload(record,label,c){
  const p=path.join(dir,c.id+'-'+label+'.tap')
  if(record.log!==p||!record.logSHA256||g.sha(fs.readFileSync(p))!==record.logSHA256)g.fail('TAP path/hash drift')
  const commandPath=path.join(dir,c.id+'-'+label+'.command.json'),command=fs.readFileSync(commandPath)
  if(g.sha(command)!==record.commandSHA256)g.fail('Command log hash drift')
  const parsed=JSON.parse(command);if(parsed.executable!==process.execPath||parsed.cwd!==g.ROOT||JSON.stringify(parsed.args)!==JSON.stringify(c.command.slice(1)))g.fail('Command differs from pinned exact selection')
  const text=fs.readFileSync(p,'utf8');return {...record,...parseTAP(text),text}
 }
 for(let i=0;i<ctx.controls.length;i++){
  const c=ctx.controls[i],r=records[i]
  if(r.id!==c.id||r.phase!=='transfer'||r.verified!==true||r.sourceBefore!==c.sourceSha256||r.sourceFault!==c.mutatedSha256||r.actualSourceFault!==c.mutatedSha256||r.sourceRestored!==c.sourceSha256||r.byteExactRestored!==true)g.fail('Source/verification receipt mismatch: '+c.id)
  for(const readset of ctx.pins.readset)if(r.readsetBefore?.[readset.file]!==readset.sha||r.readsetAfter?.[readset.file]!==readset.sha)g.fail('Readset proof mismatch: '+c.id)
  if(!clean(c,reload(r.before,'before',c))||!intended(c,reload(r.fault,'fault',c),'transfer').accepted||!clean(c,reload(r.after,'after',c)))g.fail('Required clean1/red1/restored clean1 not proved: '+c.id)
 }
 return {dir,completionSHA,verified:20,logFiles:60,phase:'transfer'}
}
module.exports={parseTAP,named,clean,errorMessage,diagnosticValue,valueProof,intended,validateTransferReceipt}
