const fs=require('fs'),p=require('path'),crypto=require('crypto');const base=__dirname;
const m=JSON.parse(fs.readFileSync(p.join(base,'manifest.json'))),cs=JSON.parse(fs.readFileSync(p.join(base,'candidates.json'))),ts=require(p.join(m.root,'node_modules/typescript'));
const sha=s=>crypto.createHash('sha256').update(s).digest('hex');
function sites(file,s){const sf=ts.createSourceFile(file,s,ts.ScriptTarget.Latest,true);if(sf.parseDiagnostics.length)throw Error('parse '+file);let a=[];function v(n){if(ts.isCallExpression(n)&&n.expression.getText(sf)==='test'&&n.arguments.length>=2)a.push({name:n.arguments[0].text??n.arguments[0].getText(sf),body:n.getText(sf)});ts.forEachChild(n,v)}v(sf);return a;}
if(process.cwd()!==m.root||fs.realpathSync(process.cwd())!==m.root)throw Error('root');
let counts={},unchanged=0;for(const g of m.readset)if(sha(fs.readFileSync(p.join(m.root,g.file)))!==g.sha256)throw Error('readset drift '+g.file);
for(const [phase,files]of Object.entries(m.phases)){let n=0;for(const [file,r]of Object.entries(files)){let s=fs.readFileSync(r.path,'utf8');if(sha(s)!==r.sha256)throw Error('snapshot');let ss=sites(file,s);n+=ss.length;let b=fs.readFileSync(m.phases.before[file].path,'utf8'),expected=b;for(const c of cs.filter(c=>c.file===file)){if(phase!=='before')expected=expected.replace(c.keeper.body,c.keeperAfter);if(phase==='cut')expected=expected.replace(c.donor.body+'\n\n','');}if(s!==expected)throw Error('full reconstruction');if(phase==='cut'){for(const old of sites(file,b)){let c=cs.find(c=>c.file===file&&(old.name===c.donor.name||old.name===c.keeper.name));if(c)continue;let curr=ss.find(x=>x.name===old.name);if(!curr||curr.body!==old.body)throw Error('untargeted');unchanged++;}}}if(n!==m.declarations[phase])throw Error('count');counts[phase]=n;}
if(unchanged!==73)throw Error('untouched');
let controls=JSON.parse(fs.readFileSync(p.join(base,'controls.json')));for(const c of controls){let source=fs.readFileSync(p.join(m.root,c.owner),'utf8'),sf=ts.createSourceFile(c.owner,source,ts.ScriptTarget.Latest,true),fn=sf.statements.find(n=>ts.isFunctionDeclaration(n)&&n.name?.text===c.scope.name),body=fn?.getText(sf);if(!body||body.split(c.from).length!==2||sha(source)!==c.ownerSha256)throw Error('control anchor');let mutant=source.slice(0,fn.getStart(sf))+body.replace(c.from,c.to)+source.slice(fn.end);if(sha(mutant)!==c.mutantSha256||ts.createSourceFile(c.owner,mutant,ts.ScriptTarget.Latest,true).parseDiagnostics.length)throw Error('control parse');}
console.log(JSON.stringify({ok:true,counts,phaseSnapshots:24,unchangedRetainedCallbacks:unchanged,changedKeeperCallbacks:1,controls:controls.length,readsetGuards:m.readset.length,ownerExecution:false,repositoryWrites:false,runtimeProof:false},null,2));
